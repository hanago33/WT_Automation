```python
# wt_image_matching.py 片段
# Portions derived from Airtest (https://github.com/AirtestProject/Airtest), Apache License 2.0
"""两段式置信度：灰度定位只负责"找位置"，最终分数一律由本模块重新计算。

要点（详见 docs/04-调研与对标/Airtest深度对标_20261002/ 卷二第 3 节）：
- 锚定像素：TM_CCOEFF_NORMED 会做均值归一化，两张近全同图上的微小差异会被
  放大成分数大波动；人为塞入 0/255 两个极端值把归一化动态范围锚定在全区间。
- HSV 三通道取 min：一票否决制，形状相同颜色不同的误报在 H 通道直接现形。
"""
import cv2
import numpy as np


def _rgb_to_gray(img: np.ndarray) -> np.ndarray:
    return cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)


def cal_ccoeff_confidence(im_source: np.ndarray, im_search: np.ndarray) -> float:
    """同尺寸灰度相似度（TM_CCOEFF_NORMED），带锚定与边界外扩。"""
    im_source = cv2.copyMakeBorder(im_source, 10, 10, 10, 10, cv2.BORDER_REPLICATE)
    im_source[0, 0] = 0
    im_source[0, 1] = 255
    src_gray = _rgb_to_gray(im_source)
    sch_gray = _rgb_to_gray(im_search)
    res = cv2.matchTemplate(src_gray, sch_gray, cv2.TM_CCOEFF_NORMED)
    return float(cv2.minMaxLoc(res)[1])


def cal_rgb_confidence(img_src_rgb: np.ndarray, img_sch_rgb: np.ndarray) -> float:
    """同尺寸彩色相似度：clip→HSV→三通道分别 matchTemplate→取最小（一票否决）。"""
    img_src_rgb = np.clip(img_src_rgb, 10, 245)
    img_sch_rgb = np.clip(img_sch_rgb, 10, 245)
    img_src_rgb = cv2.cvtColor(img_src_rgb, cv2.COLOR_BGR2HSV)
    img_sch_rgb = cv2.cvtColor(img_sch_rgb, cv2.COLOR_BGR2HSV)
    img_src_rgb = cv2.copyMakeBorder(img_src_rgb, 10, 10, 10, 10, cv2.BORDER_REPLICATE)
    img_src_rgb[0, 0] = 0
    img_src_rgb[0, 1] = 255
    src_channels = cv2.split(img_src_rgb)
    sch_channels = cv2.split(img_sch_rgb)
    channel_confidences = []
    for src_ch, sch_ch in zip(src_channels, sch_channels):
        res = cv2.matchTemplate(src_ch, sch_ch, cv2.TM_CCOEFF_NORMED)
        channel_confidences.append(float(cv2.minMaxLoc(res)[1]))
    return min(channel_confidences)
```

## 附录 B：多尺度定位内核 `multiscale_locate`（示意骨架，~150 行）

```python
# wt_image_matching.py 片段
# Portions derived from Airtest (https://github.com/AirtestProject/Airtest), Apache License 2.0
"""多尺度模板定位：比例范围预测 + 搜索区裁剪 + 达阈值早退 + 两段式复验。

与 wt_projection_helpers._match_template_multiscale 的关系：
  本函数为替换内核；无 capture_resolution 时自动退回"固定 7 档"兼容路径。
"""
import time
from dataclasses import dataclass, field

import cv2
import numpy as np

LEGACY_SCALES = [1.0, 1.1, 0.9, 1.2, 0.8, 1.35, 0.7]   # 兼容路径（与旧实现一致）
TIME_BUDGET_SECONDS = 1.0
PREDICT_MIN_RADIUS = 150          # Airtest MultiScaleTemplateMatchingPre.DEVIATION
TEMPLATE_MIN_EDGE = 10


@dataclass
class MatchResult:
    center: tuple                 # (x, y) 屏幕绝对坐标
    rectangle: tuple              # 四角 (左上,左下,右下,右上)
    confidence: float             # 两段式复验后的最终置信度
    scale_used: float             # 命中缩放比
    method: str                   # "mstpl" / "legacy7"
    elapsed_ms: float
    gray_max_val: float = 0.0     # 灰度定位阶段的原始分数（诊断用）


def _read_template_bgr(template_path: str) -> np.ndarray:
    # 中文路径安全读取（与 wt_projection_helpers 现方案/Airtest aircv.imread 同源）
    data = np.fromfile(template_path, dtype=np.uint8)
    img = cv2.imdecode(data, cv2.IMREAD_COLOR)
    if img is None:
        raise RuntimeError(f"无法读取模板图片: {template_path}")
    return img


def _predicted_ratio_range(capture_wh, screen_wh, templ_shape, scale_step):
    """Airtest _get_ratio_scope：由捕获/当前分辨率比值推导缩放比上下限。"""
    cap_w, cap_h = capture_wh
    cur_w, cur_h = screen_wh
    th, tw = templ_shape[:2]
    r_min_scale = min(cur_h / cap_h, cur_w / cap_w)
    r_max_scale = max(cur_h / cap_h, cur_w / cap_w)
    base = max(th / cur_h, tw / cur_w)
    return max(base * r_min_scale, scale_step), min(base * r_max_scale, 0.99)


def _predict_search_area(region_hint, capture_wh, templ_shape, screen_wh):
    """Airtest _get_area_scope 的 WT 版：用捕获时 region（而非 record_pos）预测搜索区。

    region_hint 是捕获时模板在截图中的位置；按捕获→当前分辨率比例映射到当前屏幕，
    半径取 max(模板当前应有尺寸, 150px)。返回 (area, scale_factor)。
    """
    cap_w, cap_h = capture_wh
    cur_w, cur_h = screen_wh
    fx, fy = cur_w / cap_w, cur_h / cap_h
    x = (region_hint["x"] + region_hint["w"] / 2) * fx
    y = (region_hint["y"] + region_hint["h"] / 2) * fy
    th, tw = templ_shape[:2]
    radius_x = max(int(tw * fx), PREDICT_MIN_RADIUS)
    radius_y = max(int(th * fy), PREDICT_MIN_RADIUS)
    area = (max(int(x - radius_x), 0), max(int(y - radius_y), 0),
            min(int(x + radius_x), cur_w), min(int(y + radius_y), cur_h))
    return area


def multiscale_locate(screen_bgr, template_path, threshold, *,
                      rgb=False, scale_step=0.005, scale_max=800,
                      capture_resolution=None, region_hint=None,
                      caller_region=None, time_budget=TIME_BUDGET_SECONDS):
    """在 screen_bgr 中定位模板。返回 MatchResult 或 None。

    caller_region: 调用方已限定的 ROI（旧签名 region 语义），坐标已含屏幕偏移。
    """
    started = time.time()
    template_bgr = _read_template_bgr(template_path)
    screen_gray = cv2.cvtColor(screen_bgr, cv2.COLOR_BGR2GRAY)
    sh, sw = screen_gray.shape[:2]
    th, tw = template_bgr.shape[:2]

    # 搜索区裁剪：caller_region（调用方显式 ROI）> region_hint（索引预测区）> 全屏
    offset_x = offset_y = 0
    work_gray = screen_gray
    work_bgr = screen_bgr
    if caller_region:
        rx, ry, rw, rh = caller_region
        work_gray = screen_gray[ry:ry + rh, rx:rx + rw]
        work_bgr = screen_bgr[ry:ry + rh, rx:rx + rw]
        sh, sw = work_gray.shape[:2]
        offset_x, offset_y = rx, ry
    elif region_hint and capture_resolution:
        area = _predict_search_area(region_hint, capture_resolution,
                                    template_bgr.shape, (sw, sh))
        offset_x, offset_y = area[0], area[1]
        work_gray = screen_gray[area[1]:area[3], area[0]:area[2]]
        work_bgr = screen_bgr[area[1]:area[3], area[0]:area[2]]
        sh, sw = work_gray.shape[:2]

    # 缩放比序列：有 capture_resolution 走预测区间，否则走兼容 7 档
    if capture_resolution:
        r_min, r_max = _predicted_ratio_range(
            capture_resolution, (sw, sh), template_bgr.shape, scale_step)
        scales = np.arange(r_min, r_max + scale_step / 2, scale_step)
        method = "mstpl"
    else:
        scales = LEGACY_SCALES
        method = "legacy7"

    best = None          # (gray_val, scale, loc, w, h)
    for scale in scales:
        w, h = max(int(round(tw * scale)), 2), max(int(round(th * scale)), 2)
        if min(w, h) < TEMPLATE_MIN_EDGE or w > sw or h > sh:
            continue
        resized = cv2.resize(template_bgr, (w, h), interpolation=cv2.INTER_AREA)
        resized_gray = cv2.cvtColor(resized, cv2.COLOR_BGR2GRAY)
        result = cv2.matchTemplate(work_gray, resized_gray, cv2.TM_CCOEFF_NORMED)
        _, max_val, _, max_loc = cv2.minMaxLoc(result)
        if best is None or max_val > best[0]:
            best = (max_val, scale, max_loc, w, h)
        # 达阈值即复验早退（Airtest multiscale_template_matching.py:141-146）
        if max_val >= threshold:
            # 达阈值即复验早退。WT 增强：Airtest 原版语义为"时间预算到点且当前轮
            # 达阈值"才退（multiscale_template_matching.py:141-146），这里改为
            # 更激进的"达阈值即退"；若想完全对齐 Airtest，把本 if 改为
            # `if time.time() - started > time_budget and max_val >= threshold:`
            verified = _verify(work_bgr, template_bgr, max_loc, w, h, rgb)
            if verified is not None and verified >= threshold:
                return _build_result(verified, max_val, scale, max_loc, w, h,
                                     offset_x, offset_y, method,
                                     (time.time() - started) * 1000)
        if time.time() - started > time_budget:
            break       # 时间预算到点，不再扫剩余比例（走全局最优复验兜底）

    # 全扫完无早退：用全局最优复验（保持"尽力而为"）
    if best is None:
        return None
    max_val, scale, max_loc, w, h = best
    verified = _verify(work_bgr, template_bgr, max_loc, w, h, rgb)
    elapsed = (time.time() - started) * 1000
    if verified is not None and verified >= threshold:
        return _build_result(verified, max_val, scale, max_loc, w, h,
                             offset_x, offset_y, method, elapsed)
    return None          # 调用方可在异常路径引用 last_best 做诊断（见附录 C 扩展点）


def _verify(work_bgr, template_bgr, max_loc, w, h, rgb):
    """两段式复验：从原彩图裁命中区→缩放回模板尺寸→重新算置信度。"""
    crop = work_bgr[max_loc[1]:max_loc[1] + h, max_loc[0]:max_loc[0] + w]
    if crop.size == 0:
        return None
    crop = cv2.resize(crop, (template_bgr.shape[1], template_bgr.shape[0]))
    return (cal_rgb_confidence(crop, template_bgr) if rgb
            else cal_ccoeff_confidence(crop, template_bgr))


def _build_result(confidence, gray_val, scale, loc, w, h, ox, oy, method, elapsed):
    x, y = loc
    lt = (x + ox, y + oy)
    rect = (lt, (x + ox, y + oy + h), (x + ox + w, y + oy + h), (x + ox + w, y + oy))
    center = (int(x + ox + w / 2), int(y + oy + h / 2))
    return MatchResult(center=center, rectangle=rect, confidence=round(confidence, 4),
                       scale_used=round(float(scale), 4), method=method,
                       elapsed_ms=round(elapsed, 1), gray_max_val=round(float(gray_val), 4))
```

> 审读提示：骨架中的早退采用 WT 增强语义（达阈值即退），代码注释里给了与 Airtest 原版严格对齐的写法；时间预算到点时走 break → 全局最优复验兜底，两者组合覆盖 Airtest 的全部行为。`_make_colorful_patch` / `_save` / `_swap_hue` 为测试辅助函数，落地时按附录 F 的构造思路补齐。
>
> **卷十实验后的修订建议（2026-10-02）**：`multiscale_locate` 签名增加 `refine_steps: int = 8`——早退命中后在该 scale ±`refine_steps*step` 范围内补扫取局部最优再复验返回。实测依据（卷十 E3/E4）：纯早退置信度贴线（0.71）而细化后恢复 1.0，中心偏差 ≤1px，代价 40→92ms（仍比 7 档盲扫快 2.6x）。点击容差大的场景可传 `refine_steps=0` 关闭。

```python
class TemplateNotFoundError(RuntimeError):
    """模板在超时内未命中。携带最后一次匹配诊断信息。"""

    def __init__(self, message, last_best=None, threshold=None):
        super().__init__(message)
        self.last_best = last_best        # MatchResult 或 None（灰度最优，供排障）
        self.threshold = threshold


def wait_template(template_key, timeout_seconds=8, interval=0.5,
                  intervalfunc=None, confidence=None, rgb=None, region=None):
    """轮询等待模板出现（Airtest loop_find 语义，cv.py:40-82）。

    intervalfunc：每轮失败后的钩子——防休眠心跳 / 弹窗处理 / 刷新动作的挂点
    （对接影刃对标 P0-5）。返回 MatchResult；超时抛 TemplateNotFoundError 并
    自动保存调试截图（成功路径不截图，避免与现有日志量冲突——如需成功截图
    参照 Airtest try_log_screen 加开关参数）。
    """
    params = image_template_index.get_template_params(template_key)   # 阶段 A 产物
    threshold = confidence if confidence is not None else params.get("threshold", 0.8)
    use_rgb = rgb if rgb is not None else bool(params.get("rgb", False))
    template_path = get_template_path(template_key)
    if not os.path.exists(template_path):
        raise FileNotFoundError(f"缺少图片模板: {template_path}")

    deadline = time.time() + timeout_seconds
    last_best = None
    while time.time() < deadline:
        screen = _grab_screen_bgr(region)                     # 单次截图，多尺度复用
        result = wt_image_matching.multiscale_locate(
            screen, template_path, threshold,

            capture_resolution=params.get("capture_resolution"),
            region_hint=params.get("region"),
            caller_region=region)
        if result is not None:
            return result
        last_best = wt_image_matching.last_best_result        # 内核暴露的诊断信息
        if intervalfunc is not None:
            try:
                intervalfunc()
            except Exception as exc:                          # 钩子异常不中断等待
                _LOG_STEP(f"wait_template.intervalfunc 异常: {exc}")
        time.sleep(interval)

    capture_debug_screenshot(f"template_miss_{_safe_log_filename_component(template_key)}")

        last_best=last_best, threshold=threshold)
```

```python
"""扫描码键盘通道：SendInput + KEYEVENTF_SCANCODE，绕过虚拟键映射层。

适用：只认 DirectInput 扫描码的程序（部分专业软件/游戏）。完整键位表
（84 标准键 + 17 扩展键）见 Airtest core/win/ctypesinput.py:10-119，
落地时按 WT 实际需要的键位子集引入。
"""
import ctypes
import win32con

KEYEVENTF_SCANCODE = 0x0008
_SCAN = {"ENTER": 0x1C, "LCTRL": 0x1D, "A": 0x1E, "V": 0x2F, "ESCAPE": 0x01, ...}


def _send_scan(scan_code: int, flags: int) -> None:
    class KeyBdInput(ctypes.Structure):
        _fields_ = [("wVk", ctypes.c_ushort), ("wScan", ctypes.c_ushort),
                    ("dwFlags", ctypes.c_ulong), ("time", ctypes.c_ulong),
                    ("dwExtraInfo", ctypes.POINTER(ctypes.c_ulong))]
    class Input(ctypes.Structure):
        _fields_ = [("type", ctypes.c_ulong), ("ki", KeyBdInput)]
    inp = Input(type=ctypes.c_ulong(win32con.INPUT_KEYBOARD),
                ki=KeyBdInput(0, scan_code, flags, 0, None))
    ctypes.windll.user32.SendInput(1, ctypes.byref(inp), ctypes.sizeof(inp))


def key_press_scan(key: str) -> None:
    _send_scan(_SCAN[key.upper()], KEYEVENTF_SCANCODE)


def key_release_scan(key: str) -> None:
    _send_scan(_SCAN[key.upper()], KEYEVENTF_SCANCODE | win32con.KEYEVENTF_KEYUP)
```
