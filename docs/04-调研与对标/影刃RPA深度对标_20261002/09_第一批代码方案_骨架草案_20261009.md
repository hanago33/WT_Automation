### 1.2 新增：`tools/generate_command_catalog.py`（完整骨架）

```python
# encoding: utf-8
"""生成 Agent 侧指令总目录（command_catalog.json）。

单一事实源：根目录 wt_action_schema.py + wt_action_defaults.py（本脚本只读取，不修改）。
产物：WT_AUTOMATION_Agent/command_catalog.json —— 供 Agent 提示词装配与一致性测试消费。

用法（项目根执行）：
    python tools/generate_command_catalog.py           # 生成/刷新产物
    python tools/generate_command_catalog.py --check   # 只校验产物是否与源码一致（不一致退出码 1）
"""
import argparse
import hashlib
import json
import os
import sys

PROJECT_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
if PROJECT_ROOT not in sys.path:
    sys.path.insert(0, PROJECT_ROOT)

from wt_action_schema import (          # noqa: E402
    ACTION_SCHEMAS,
    ALLOWED_CONTINUE_WHEN_CONDITIONS,
    ALLOWED_ON_ERROR_MODES,
    ALLOWED_PARENT_WINDOW_FRAMEWORK_IDS,
    ALLOWED_RELATIVE_REGION_ANCHORS,
    STEP_POLICY_ON_FAIL_MODES,
)
from wt_action_defaults import ACTION_DEFAULT_CONFIGS  # noqa: E402

CATALOG_PATH = os.path.join(PROJECT_ROOT, "WT_AUTOMATION_Agent", "command_catalog.json")

# 逐字段复制（ACTION_SCHEMAS 是 dict：迭代它得到的是键名，必须显式取键——
# 卷四示意代码 `for s in ACTION_SCHEMAS` 即在此处出错）
_SCHEMA_FIELDS = (
    "label", "description", "target_required", "input_required",
    "input_key", "input_label", "show_timeout", "suggested_columns",
)


def build_catalog() -> dict:
    actions = []
    for name in sorted(ACTION_SCHEMAS):
        schema = ACTION_SCHEMAS[name]
        item = {"action": name}
        for field in _SCHEMA_FIELDS:
            if field in schema:
                value = schema[field]
                item[field] = list(value) if isinstance(value, tuple) else value
        item["defaults"] = dict(ACTION_DEFAULT_CONFIGS.get(name, {}))
        actions.append(item)
    payload = {
        "version": 1,
        "sourceModule": "wt_action_schema",
        "sourceHash": _source_hash(actions),
        "actions": actions,
        "enums": {
            "onError": list(ALLOWED_ON_ERROR_MODES),
            "continueWhen": list(ALLOWED_CONTINUE_WHEN_CONDITIONS),
            "relativeRegionAnchors": list(ALLOWED_RELATIVE_REGION_ANCHORS),
            "parentWindowFrameworkIds": list(ALLOWED_PARENT_WINDOW_FRAMEWORK_IDS),
            "stepPolicyOnFail": list(STEP_POLICY_ON_FAIL_MODES),
        },
    }
    return payload


def _source_hash(actions: list) -> str:
    canonical = json.dumps(actions, ensure_ascii=False, sort_keys=True)
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()[:16]


def write_catalog(payload: dict, path: str = CATALOG_PATH) -> str:
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w", encoding="utf-8") as file_obj:
        json.dump(payload, file_obj, ensure_ascii=False, indent=2)
        file_obj.write("\n")
    return path


def main(argv=None) -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--check", action="store_true", help="只校验产物与源码一致（不写盘）")
    args = parser.parse_args(argv)
    payload = build_catalog()
    if args.check:
        try:
            with open(CATALOG_PATH, "r", encoding="utf-8") as file_obj:
                on_disk = json.load(file_obj)
        except (OSError, json.JSONDecodeError):
            print("catalog 不存在或不可解析，请先运行生成器")
            return 1
        if on_disk != payload:
            print("catalog 与 wt_action_schema 不一致，请重新生成")
            return 1
        print("catalog 与源码一致")
        return 0
    write_catalog(payload)
    print("written", CATALOG_PATH)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

import os

                action=item.get("action", ""),
                label=item.get("label", ""),
                description=item.get("description", ""),

    enums = payload.get("enums", {})
    lines.append("")
    lines.append("等待条件枚举 continueWhen: " + "、".join(enums.get("continueWhen", [])))
    lines.append("失败策略枚举 onError: " + "、".join(enums.get("onError", [])))

### 2.1 新增：`wt_run_guards.py`（完整骨架）

```python
# encoding: utf-8
"""运行期保障：防休眠 / 会话告警 / 卡死登记。

对标：
- [影刃] os_sleep.start_prevent_os_sleep（SetThreadExecutionState 常驻）→ PreventSleep；
- [影刃] StuckChecker（60s 看门狗 + 主线程栈转储）→ StuckRegistry（WT 化：不杀进程，只落 JSONL 证据）；
- 会话检测复用 wt_task_queue.check_desktop_interactive（延迟 import，避免模块级依赖）。

零第三方依赖（ctypes + 标准库）；可被 headless 流程导入（不 import tkinter）。
"""
import ctypes
import sys
import threading
import time
import traceback

import wt_logging

ES_CONTINUOUS = 0x80000000
ES_SYSTEM_REQUIRED = 0x00000001

STUCK_DEFAULT_INTERVAL = 60.0


class PreventSleep:
    """运行期间阻止系统休眠（不阻止熄屏）。

    用法（在 run_automation 内手工 start/stop，避免重排大函数缩进）：
        guard = PreventSleep(); guard.start()   # try 前定义、try 内 start
        try: ...
        finally: guard.stop()
    """

    def __init__(self):
        self._active = False

    def start(self) -> bool:
        if self._active:
            return True
        try:
            ctypes.windll.kernel32.SetThreadExecutionState(ES_CONTINUOUS | ES_SYSTEM_REQUIRED)
            self._active = True
        except Exception:          # 非 Windows / API 不可用：静默降级
            self._active = False
        return self._active

    def stop(self) -> None:
        if not self._active:
            return
        try:
            ctypes.windll.kernel32.SetThreadExecutionState(ES_CONTINUOUS)
        except Exception:
            pass
        self._active = False


def check_session_alert():
    """运行前会话告警：锁屏 / Session 0 时写 WARN 事件（不阻断流程）。

    返回复用 wt_task_queue.check_desktop_interactive（失败返回 True，绝不因检测失败阻断）。
    依赖方向说明：WT_AUT_recorded 已模块级 import wt_task_queue（队列路径常量），无新增负担。
    """
    try:
        from wt_task_queue import check_desktop_interactive
        ok, reason = check_desktop_interactive()
    except Exception as exc:
        return True, "检测跳过 ({})".format(exc)
    if not ok:
        wt_logging.log_event(
            wt_logging.WARN, "运行环境告警: {}".format(reason), event="session_alert",
        )
    return ok, reason


class StuckRegistry:
    """步骤级卡死登记：step_start/step_done + 看门狗周期巡检。

    超时未结束 → 向 JSONL 写 stuck_suspect 事件（含**主线程**栈），不中断流程
    （单进程架构下只留证据；主线程栈经 sys._current_frames() 获取——在巡检线程里
    traceback.format_stack() 拿到的是巡检线程自己的栈，这是关键实现点）。
    """

    def __init__(self, interval=STUCK_DEFAULT_INTERVAL, sink=None):
        self._interval = float(interval)
        self._sink = sink or self._default_sink
        self._cur = None
        self._t0 = 0.0
        self._lock = threading.Lock()
        self._stop = threading.Event()
        self._thread = None
        self._main_thread_id = None

    def start(self) -> None:
        if self._thread is not None:
            return
        self._main_thread_id = threading.main_thread().ident
        self._thread = threading.Thread(
            target=self._routine, daemon=True, name="wt-stuck-watchdog",
        )
        self._thread.start()

    def stop(self) -> None:
        self._stop.set()

    def step_start(self, step_id) -> None:
        with self._lock:
            self._cur, self._t0 = str(step_id or ""), time.monotonic()

    def step_done(self, step_id) -> None:
        with self._lock:
            if self._cur == str(step_id or ""):
                self._cur = None

    def _main_thread_stack(self):
        try:
            frame = sys._current_frames().get(self._main_thread_id)
            return traceback.format_stack(frame, limit=40) if frame is not None else []
        except Exception:
            return []

    @staticmethod
    def _default_sink(payload):
        wt_logging.log_event(
            wt_logging.WARN,
            payload.get("message", "步骤疑似卡死"),
            event="stuck_suspect",
            step_id=payload.get("step", ""),
            elapsed_ms=payload.get("elapsedMs"),
            detail={"mainStack": payload.get("mainStack", [])},
        )

    def _routine(self):
        while not self._stop.wait(self._interval):
            with self._lock:
                stuck_id = self._cur
                elapsed = time.monotonic() - self._t0 if stuck_id else 0.0
            if not stuck_id:
                continue
            self._sink({
                "event": "stuck_suspect",
                "step": stuck_id,
                "elapsedMs": round(elapsed * 1000.0, 1),
                "message": "步骤疑似卡死: step={}, 已运行 {:.0f}s（阈值 {:.0f}s）".format(
                    stuck_id, elapsed, self._interval
                ),
                "mainStack": self._main_thread_stack(),
            })
```

### 3.1 新增：`wt_run_reporting.py` 快照函数（完整骨架）

```python
# encoding: utf-8
# 追加到 wt_run_reporting.py（模块尾部；与 report_step_result 同文件，随报告一起演进）
# 对标：[影刃] SnapshotValueHandler inputs[{name,value,type,isFullValue}] + mask_sensitive_params；
#       [Airtest] @logwrap 的 inspect 参数快照思想；[OpenAdapt] 报告"证据强度"透明度。

_SNAPSHOT_MAX_CHARS = 5000
_SENSITIVE_KEY_HINTS = ("password", "pwd", "token", "secret", "apikey", "api_key", "credential")


def build_step_inputs_snapshot(action_config, max_chars=_SNAPSHOT_MAX_CHARS):
    """构造步骤输入快照：resolved actionConfig 的逐键投影。

    规则（与影刃一致）：
    - 键名以 `_` 开头（内部字段）跳过；
    - 敏感键掩码 `***`；
    - 长值截断保头部，isFullValue 标记是否截断（诊断时可判断证据完整性）。
    """
    items = []
    if not isinstance(action_config, dict):
        return items
    for key, value in action_config.items():
        key_text = str(key)
        if key_text.startswith("_"):
            continue
        lowered = key_text.lower()
        if any(hint in lowered for hint in _SENSITIVE_KEY_HINTS):
            items.append({"name": key_text, "value": "***", "type": "masked", "isFullValue": True})
            continue
        if isinstance(value, (dict, list)):
            text = json.dumps(value, ensure_ascii=False)   # 模块顶部已有 import json
        else:
            text = str(value)
        items.append({
            "name": key_text,
            "value": text[:max_chars],
            "type": type(value).__name__,
            "isFullValue": len(text) <= max_chars,
        })
    return items
```
